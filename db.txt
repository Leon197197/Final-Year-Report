DROP TABLE IF EXISTS public.mail_logs;
DROP TABLE IF EXISTS public.logs_analysis;
CREATE TABLE public.logs_analysis (
    id SERIAL PRIMARY KEY,
    analysis_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    llm_result VARCHAR(200),
    folder VARCHAR(200),
    mail_ReceivedTime TIMESTAMP,
    mail_sender VARCHAR(200),
    mail_subject VARCHAR(200),
    mail_body VARCHAR(1000),
    level VARCHAR(10),
    factor_sentiment VARCHAR(10),
    factor_time VARCHAR(10)
);

DROP TABLE IF EXISTS public.logs_reply;
CREATE TABLE public.logs_reply (
    id SERIAL PRIMARY KEY,
    reply_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    llm_result VARCHAR(200),
    reply_topic VARCHAR(200),
    mail_ReceivedTime TIMESTAMP,
    mail_sender VARCHAR(200),
    mail_subject VARCHAR(200),
    mail_body VARCHAR(1000)
);
